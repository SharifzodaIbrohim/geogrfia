def _iso(v) -> str:
    if v is None or v == "":
        return ""
    if isinstance(v, datetime):
        try:
            return v.strftime("%Y-%m-%d %H:%M:%S")
        except Exception:
            return str(v)[:19]
    if isinstance(v, date):
        return v.strftime("%Y-%m-%d")
    if hasattr(v, "isoformat"):
        try:
            s = v.isoformat().replace("T", " ")
            return s[:19] if len(s) >= 19 else s
        except Exception:
            pass
    s = str(v).replace("T", " ").strip()
    if len(s) >= 19 and s[4] == "-" and s[10] == " ":
        return s[:19]
    if len(s) >= 10 and s[4] == "-":
        return s[:10]
    return s

def _today() -> str:
