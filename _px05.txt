                if ci in text_cols or (isinstance(val, str) and val.isdigit() and len(val) >= 12):
                    cell = ws.cell(ri, ci, str(val))
                    cell.number_format = "@"
                else:
                    cell = ws.cell(ri, ci, val)
                cell.border = thin
        for col in range(1, len(headers) + 1):
            letter = get_column_letter(col)
            maxlen = len(str(headers[col - 1] or ""))
            for ri in range(2, min(len(rows) + 2, 40)):
                v = ws.cell(ri, col).value
                if v is not None:
                    maxlen = max(maxlen, min(len(str(v)), 40))
            ws.column_dimensions[letter].width = min(max(maxlen + 2, 10), 36)
        if rows:
            ws.auto_filter.ref = ws.dimensions
        ws.freeze_panes = "A2"
    buf = io.BytesIO()
    wb.save(buf)
    buf.seek(0)
    return Response(
        buf.getvalue(),
        mimetype="application/vnd.openxmlformats-officedocument.spreadsheetml.sheet",
        headers={"Content-Disposition": f'attachment; filename="{filename}"'},
    )
