
            # Load from olympiad_questions + olympiad_options (Neon schema)
            try:
                rows = conn.execute(text("""
                    SELECT q.id::text AS qid, q.text AS qtext,
                           COALESCE(q.qtype, 'single') AS qtype,
                           q.payload, q.sort_order AS qord,
                           o.id::text AS oid, o.text AS otext, o.is_correct, o.sort_order AS oord
                    FROM olympiad_questions q
                    LEFT JOIN olympiad_options o ON o.question_id = q.id
                    WHERE q.olympiad_id::text = :oid
                    ORDER BY q.sort_order, o.sort_order
                """), {"oid": str(att["olympiad_id"])}).mappings().all()
                by_q = {}
                for r in rows:
                    qid = r["qid"]
                    if qid not in by_q:
                        by_q[qid] = {
                            "id": qid,
                            "text": r["qtext"] or "",
                            "type": r["qtype"] or "single",
                            "options": [],
                            "answer": None,
                            "_correct_idx": None,
                        }
                        pl = r.get("payload")
                        if isinstance(pl, str):
                            try:
                                import json as _j
                                pl = _j.loads(pl)
                            except Exception:
                                pl = None
                        if isinstance(pl, dict):
                            by_q[qid].update({k: pl[k] for k in pl if k not in ("options",)})
                    if r.get("oid"):
                        by_q[qid]["options"].append({
                            "id": r["oid"],
                            "text": r["otext"] or "",
                            "isCorrect": bool(r["is_correct"]),
                        })
                        if r["is_correct"] and by_q[qid]["_correct_idx"] is None:
                            by_q[qid]["_correct_idx"] = len(by_q[qid]["options"]) - 1
                qs = list(by_q.values())
                for q in qs:
                    if q.get("_correct_idx") is not None:
                        q["answer"] = q["_correct_idx"]
                    q["options"] = [o["text"] if isinstance(o, dict) else str(o) for o in q.get("options") or []]
            except Exception as e:
                log.warning("review qs table: %s", e)
                try: conn.rollback()
                except Exception: pass

    items = []
    ok_n = bad_n = blank_n = 0
    LETTERS = "ABCDEFGHIJKLMNOPQRSTUVWXYZ"
    for i, q in enumerate(qs or []):

        opts = _opts(q)

        if aa is None and i < len(answers_rows):
            aa = answers_rows[i]
        if aa:
            student = aa.get("selected_text") or aa.get("answer_text")
            if student is None and aa.get("selected_idx") is not None:
                try:
                    si = int(aa["selected_idx"])
                    if 0 <= si < len(opts):
                        student = opts[si]
                    else:
                        student = str(si)
                except Exception:
                    student = str(aa.get("selected_idx"))
        # 2) answers_json dict
        if student is None and isinstance(answers_json, dict):
            raw = answers_json.get(qid)
            if raw is None:
                raw = answers_json.get(str(i))
            if raw is None:
                for k, v in answers_json.items():
                    if str(k) == qid or str(k).endswith(qid[-8:] if len(qid) > 8 else qid):
                        raw = v
                        break
            if raw is None and len(answers_json) == 1:
                raw = next(iter(answers_json.values()))
            student = _map_ans(raw, opts)
        # 3) answers_json list
        if student is None and isinstance(answers_json, list) and i < len(answers_json):
            student = _map_ans(answers_json[i], opts)
        is_blank = student is None or str(student).strip() in ("", "None")
        is_ok = (not is_blank) and correct is not None and str(student).strip().lower() == str(correct).strip().lower()
        if is_blank: blank_n += 1; label = "Беҷавоб"
        elif is_ok: ok_n += 1; label = "Дуруст"
        else: bad_n += 1; label = "Нодуруст"
        stu_s = "" if is_blank else str(student).strip().lower()
        cor_s = "" if correct is None else str(correct).strip().lower()
        rich = [{"letter": LETTERS[oi] if oi < 26 else str(oi+1), "text": ot,
                 "isCorrect": bool(cor_s and ot.strip().lower()==cor_s),
                 "studentSelected": bool(stu_s and ot.strip().lower()==stu_s)} for oi, ot in enumerate(opts)]
        items.append({

            "question": text_q, "text": text_q, "options": rich or opts,
            "studentAnswer": "" if is_blank else str(student),
            "correctAnswer": "" if correct is None else str(correct),
